cat /etc/X11/default-display-manager  // To see the default display manager

VNC Configration : 

apt install tigervnc-standalone-server
adduser kirin // add user
deluser kirin // del user
password:

root user : sudo su -
vncserver -localhost no // Start VNC Server
vncserver -list // List of Running VNC Servers

su - omega/kirin // Switch Between Users

vncserver -kill :1
